let sortedVersionsCache = [];
const TEMPORAL_LEVELS = { "Weekly": 0, "Monthly": 1, "Yearly": 2, "NA": 3 };
const TEMPORAL_TEXT = { 0: "Weekly", 1: "Monthly", 2: "Yearly", 3: "NA" };
const TEMPORAL_COLORSCALE = [
    [0.0, '#34D399'], [0.25, '#34D399'],
    [0.25, '#A7F3D0'], [0.5, '#A7F3D0'],

const SPATIAL_TEXT = { 0: "Admin2", 1: "Admin1", 2: "Admin0", 3: "NA" };
const SPATIAL_COLORSCALE = [
    [0.0,  '#EC4899'], [0.25, '#EC4899'],
    [0.25, '#F9A8D4'], [0.50, '#F9A8D4'],
    [0.50, '#FCE7F3'], [0.75, '#FCE7F3'],

];

function showMessage(message, type = 'info', targetDivId = 'messageBox') {
    const box = document.getElementById(targetDivId);
    if (!box) {
        console.error("showMessage: Target div not found:", targetDivId);
        return;
    }
    box.innerHTML = message; // Be cautious if 'message' can contain untrusted HTML

    let baseClasses = "";
    let typeClasses = "";

    if (targetDivId === 'summaryPanel') {
        baseClasses = 'summary-panel lg:w-[320px] xl:w-[380px] flex-shrink-0 bg-white p-6 rounded-lg shadow-lg border border-slate-200 text-slate-700';
        // Ensure these styles are applied if they get cleared
        box.style.height = '780px';
        box.style.overflowY = 'auto';
    } else {
        baseClasses = 'message-box mt-6 p-4 rounded-lg shadow-md';
        if (type === 'error') typeClasses = 'bg-red-100 border border-red-400 text-red-700';
        else if (type === 'info') typeClasses = 'bg-blue-100 border border-blue-400 text-blue-700';
        else typeClasses = 'bg-sky-100 border border-sky-400 text-sky-700'; // For 'comparison_summary', 'global_summary'
    }
    box.className = `${baseClasses} ${typeClasses}`;
    box.style.display = 'block';
}

function hideMessage(targetDivId = 'messageBox') {
    const box = document.getElementById(targetDivId);
    if (box) {
        box.style.display = 'none';
        box.innerHTML = '';
    }
}

function sortVersions(v1, v2) {
    const parsePart = (part) => String(part).replace('V', '').split('_').map(Number);
    const parts1 = parsePart(v1);
    const parts2 = parsePart(v2);
    for (let i = 0; i < Math.max(parts1.length, parts2.length); i++) {
        const p1 = parts1[i] || 0;
        const p2 = parts2[i] || 0;
        if (p1 !== p2) return p1 - p2;
    }
    return 0;
}

function loadCSVData() {
    return new Promise((resolve, reject) => {
        // "heatmap.csv" is expected to be in the same directory as the rendered HTML,
        // or accessible via a relative/absolute path from the HTML's location.
        Papa.parse("heatmap.csv", {
            download: true, // This tells PapaParse to fetch the file
            header: true,
            dynamicTyping: true,
            skipEmptyLines: true,
            complete: function(result) {
                if (result.data && result.data.length > 0) {
                    resolve(result.data);
                } else {
                    const errorMsg = result.errors && result.errors.length > 0 ? result.errors[0].message : 'No data found in CSV or CSV is empty.';
                    showMessage(`Error parsing CSV: ${errorMsg}. Ensure 'heatmap.csv' is valid and in the correct location relative to the output HTML.`, 'error');
                    reject(new Error(errorMsg));
                }
            },
            error: function(err, file) {
                // For download errors, err might be an XHR error object or similar
                let errorDetail = err ? (err.message || 'Network error or file not found') : 'Unknown error';
                if (err && err.target && err.target.status === 0) {
                     errorDetail = "File not found or Cross-Origin Request Blocked. Ensure 'heatmap.csv' is in the same directory as the HTML file, or serve files from a local web server.";
                } else if (err && err.target && err.target.status === 404) {
                     errorDetail = "File 'heatmap.csv' not found (404). Check the file path and ensure it's in the same directory as the output HTML.";
                }
                showMessage(`Error loading heatmap.csv: ${errorDetail}`, 'error');
                reject(new Error(errorDetail));
            }
        });
    });
}

function populateVersionDropdowns(allData, currentVersionSelectId, previousVersionSelectId) {
    const currentDropdown = document.getElementById(currentVersionSelectId);
    const previousDropdown = document.getElementById(previousVersionSelectId);
    currentDropdown.innerHTML = '';
    previousDropdown.innerHTML = '';

    if (!allData || allData.length === 0) {
        currentDropdown.disabled = true;
        previousDropdown.disabled = true;
        sortedVersionsCache = []; // Reset cache
        return;
    }
    sortedVersionsCache = [...new Set(allData.map(d => String(d.version)))].sort(sortVersions);
    sortedVersionsCache.forEach(v => {
        currentDropdown.add(new Option(v, v));
        previousDropdown.add(new Option(v, v));
    });

    if (sortedVersionsCache.length > 0) {
        currentDropdown.value = sortedVersionsCache[sortedVersionsCache.length - 1];
        currentDropdown.disabled = false;
        // updatePreviousVersionSelection will be called to set the previous dropdown
    } else {
        currentDropdown.disabled = true;
        previousDropdown.disabled = true;
    }
}

function updatePreviousVersionSelection() {
    const currentVersionSelect = document.getElementById('currentVersionSelect');
    const previousVersionSelect = document.getElementById('previousVersionSelect');
    const selectedCurrentVersion = currentVersionSelect.value;
    const currentIndex = sortedVersionsCache.indexOf(selectedCurrentVersion);

    previousVersionSelect.disabled = sortedVersionsCache.length <= 1 || currentIndex === 0;

    if (currentIndex > 0) {
        previousVersionSelect.value = sortedVersionsCache[currentIndex - 1];
    } else if (sortedVersionsCache.length > 0 && sortedVersionsCache.length <=1 ) {
        previousVersionSelect.value = sortedVersionsCache[0];
    } else if (sortedVersionsCache.length > 1) {
        previousVersionSelect.value = sortedVersionsCache[0];
    }
}

function updateHeatmap(region, allData, performComparison, selectedCurrentVersion, selectedPreviousVersion, selectedResolutionType) {
    hideMessage();
    hideMessage('summaryPanel');
    const heatmapDiv = document.getElementById('heatmapDiv');
    if (!heatmapDiv) {
        console.error("heatmapDiv element not found in updateHeatmap.");
        return;
    }
    Plotly.purge(heatmapDiv);


    const YEAR_MIN = 1990;
    const YEAR_MAX = 2024;

    if (!allData || !Array.isArray(allData) || allData.length === 0) {
        showMessage('No data available to display heatmap. Check data source.', 'error', 'messageBox');
        return;
    }

    const versionForHeatmap = selectedCurrentVersion;
    if (!versionForHeatmap) {
        showMessage('Please select a version to display.', 'error', 'messageBox');
        return;
    }

    let currentVersionData = allData.filter(d =>
        String(d.version) === String(versionForHeatmap) &&
        d.region === region &&
        d.Year >= YEAR_MIN && d.Year <= YEAR_MAX
    );

    if (currentVersionData.length === 0) {
        const emptyMsg = `No data found for Region: '${region}', Version: ${versionForHeatmap} (Years: ${YEAR_MIN}-${YEAR_MAX}).`;
        showMessage(emptyMsg, 'info', 'messageBox');
        let summaryText = `<h4 class="text-xl font-semibold mb-3 pb-2 border-b border-slate-300">Summary</h4><ul><li>Displaying: ${versionForHeatmap}</li><li>Region: ${region}</li><li>Years: ${YEAR_MIN}-${YEAR_MAX}</li><li>Total Countries: 0</li><li>Total Tiles: 0</li></ul>`;
        showMessage(summaryText, performComparison ? 'comparison_summary' : 'global_summary', 'summaryPanel');
        return;
    }

    const uniqueYears = [...new Set(currentVersionData.map(d => d.Year))].sort((a, b) => a - b);
    const uniqueCountries = [...new Set(currentVersionData.map(d => d.adm_0_name))].sort();

    if (uniqueCountries.length === 0 || uniqueYears.length === 0) {
        showMessage(`Not enough distinct data (countries or years) for ${region}, Version: ${versionForHeatmap}.`, 'info', 'messageBox');
        return;
    }

    const zValuesMatrix = uniqueCountries.map(() => uniqueYears.map(() => (selectedResolutionType === "Temporal" ? TEMPORAL_LEVELS["NA"] : SPATIAL_LEVELS["NA"])));
    const hoverTextMatrix = uniqueCountries.map(() => uniqueYears.map(() => ''));

    let counts = {};
    const levelTexts = selectedResolutionType === "Temporal" ? TEMPORAL_TEXT : SPATIAL_TEXT;
    Object.values(levelTexts).forEach(text => counts[text] = 0);
    let totalDisplayedTiles = 0;

    currentVersionData.forEach(d => {
        const yearIndex = uniqueYears.indexOf(d.Year);
        const countryIndex = uniqueCountries.indexOf(d.adm_0_name);
        if (yearIndex !== -1 && countryIndex !== -1) {
            totalDisplayedTiles++;
            let resValue;
            const currentTResText = d.T_res_nm || "NA";
            const currentSResText = d.S_res_nm || "NA";

            if (selectedResolutionType === "Temporal") {
                resValue = TEMPORAL_LEVELS[currentTResText] !== undefined ? TEMPORAL_LEVELS[currentTResText] : TEMPORAL_LEVELS["NA"];
                counts[currentTResText] = (counts[currentTResText] || 0) + 1;
            } else {
                resValue = SPATIAL_LEVELS[currentSResText] !== undefined ? SPATIAL_LEVELS[currentSResText] : SPATIAL_LEVELS["NA"];
                counts[currentSResText] = (counts[currentSResText] || 0) + 1;
            }
            zValuesMatrix[countryIndex][yearIndex] = resValue;
            hoverTextMatrix[countryIndex][yearIndex] = `<b>Country:</b> ${uniqueCountries[countryIndex]}<br><b>Year:</b> ${uniqueYears[yearIndex]}<br><b>T.Res:</b> ${currentTResText}<br><b>S.Res:</b> ${currentSResText}`;
        }
    });

    const plotData = [{
        x: uniqueYears, y: uniqueCountries, z: zValuesMatrix, type: 'heatmap',
        text: hoverTextMatrix,
        hovertemplate: '%{text}<extra></extra>',
        zmin: 0,
        zmax: selectedResolutionType === "Temporal" ? Object.keys(TEMPORAL_LEVELS).length - 1 : Object.keys(SPATIAL_LEVELS).length - 1,
        colorscale: selectedResolutionType === "Temporal" ? TEMPORAL_COLORSCALE : SPATIAL_COLORSCALE,
        showscale: true,
        colorbar: {
            tickvals: Object.values(selectedResolutionType === "Temporal" ? TEMPORAL_LEVELS : SPATIAL_LEVELS),
            ticktext: Object.keys(selectedResolutionType === "Temporal" ? TEMPORAL_LEVELS : SPATIAL_LEVELS),
            len: 0.7, y: 0.5, yanchor: 'middle', thickness: 15,
            tickfont: { family: 'Inter, sans-serif', size: 10, color: '#334155'},
            title: { text: selectedResolutionType === "Temporal" ? 'Temporal Res.' : 'Spatial Res.', font: { family: 'Inter, sans-serif', size: 12, color: '#334155'}, side: 'right'}
        },
        xgap: 1, ygap: 1
    }];

    const summaryPanelDiv = document.getElementById('summaryPanel');
    if (summaryPanelDiv) summaryPanelDiv.style.display = 'block';

    if (performComparison && selectedCurrentVersion && selectedPreviousVersion && selectedCurrentVersion !== selectedPreviousVersion) {
        let comparisonSummaryText = `<h4 class="text-xl font-semibold text-slate-800 mb-4 pb-2 border-b border-slate-300">Comparison Summary</h4><ul class="space-y-1 text-sm text-slate-700">`;
        comparisonSummaryText += `<li><strong>Displaying:</strong> ${selectedCurrentVersion}</li>`;
        comparisonSummaryText += `<li><strong>Comparing with:</strong> ${selectedPreviousVersion}</li>`;

        const prevVersionDataFull = allData.filter(d => String(d.version) === String(selectedPreviousVersion) && d.Year >= YEAR_MIN && d.Year <= YEAR_MAX);
        const prevCountriesAll = new Set(prevVersionDataFull.map(d => d.adm_0_name));
        const newCountriesList = uniqueCountries.filter(c => !prevCountriesAll.has(c));

        const prevVersionDataRegion = prevVersionDataFull.filter(d => d.region === region);
        const prevDataMap = new Map();
        prevVersionDataRegion.forEach(d_prev => {
            prevDataMap.set(`${d_prev.adm_0_name}_${d_prev.Year}`, {
                t_level: TEMPORAL_LEVELS[d_prev.T_res_nm || "NA"],
                s_level: SPATIAL_LEVELS[d_prev.S_res_nm || "NA"],
            });
        });

        let newMeaningfulForSelectedTypeCount = 0;
        let improvedForSelectedTypeCount = 0;
        const zNewHighlight = uniqueCountries.map(() => uniqueYears.map(() => null));
        const zImprovedHighlight = uniqueCountries.map(() => uniqueYears.map(() => null));

        currentVersionData.forEach(d_curr => {
            const tileKey = `${d_curr.adm_0_name}_${d_curr.Year}`;
            const prevEntry = prevDataMap.get(tileKey);
            const curr_t_level = TEMPORAL_LEVELS[d_curr.T_res_nm || "NA"];
            const curr_s_level = SPATIAL_LEVELS[d_curr.S_res_nm || "NA"];
            const countryIndex = uniqueCountries.indexOf(d_curr.adm_0_name);
            const yearIndex = uniqueYears.indexOf(d_curr.Year);
            let classifiedAsNewMeaningfulPrimary = false;
            let classifiedAsImprovedPrimary = false;
            const currentPrimaryLevel = (selectedResolutionType === "Temporal") ? curr_t_level : curr_s_level;
            const NA_LEVEL_FOR_PRIMARY = (selectedResolutionType === "Temporal") ? TEMPORAL_LEVELS["NA"] : SPATIAL_LEVELS["NA"];
            const currentPrimaryIsNA = currentPrimaryLevel === NA_LEVEL_FOR_PRIMARY;
            const previousPrimaryLevel = prevEntry ?
                                      ((selectedResolutionType === "Temporal") ? prevEntry.t_level : prevEntry.s_level)
                                      : undefined;

            if (!currentPrimaryIsNA && (!prevEntry || previousPrimaryLevel === NA_LEVEL_FOR_PRIMARY)) {
                newMeaningfulForSelectedTypeCount++;
                classifiedAsNewMeaningfulPrimary = true;
            } else if (prevEntry && !currentPrimaryIsNA &&
                       previousPrimaryLevel !== NA_LEVEL_FOR_PRIMARY &&
                       currentPrimaryLevel < previousPrimaryLevel) {
                improvedForSelectedTypeCount++;
                classifiedAsImprovedPrimary = true;
            }

            if (countryIndex !== -1 && yearIndex !== -1) {
                let hoverPrefix = "";
                if (classifiedAsNewMeaningfulPrimary) {
                    zNewHighlight[countryIndex][yearIndex] = 4;
                    hoverPrefix = `<span style="color:#DC2626; font-weight:bold;">New Meaningful ${selectedResolutionType === "Temporal" ? "T.Res" : "S.Res"}</span><br>`;
                } else if (classifiedAsImprovedPrimary) {
                    zImprovedHighlight[countryIndex][yearIndex] = 5;
                    hoverPrefix = `<span style="color:#2563EB; font-weight:bold;">Improved ${selectedResolutionType === "Temporal" ? "T.Res" : "S.Res"}</span><br>`;
                }
                if (hoverPrefix) {
                    // Update the main plot's hover text matrix directly
                    plotData[0].text[countryIndex][yearIndex] = hoverPrefix + plotData[0].text[countryIndex][yearIndex];
                }
            }
        });

        comparisonSummaryText += `<li>New countries in region: <span class="summary-new-country">${newCountriesList.length}</span> ${newCountriesList.length > 0 ? `(${newCountriesList.join(', ')})` : ''}</li>`;
        comparisonSummaryText += `<li>New meaningful tiles (for ${selectedResolutionType}): <span class="summary-new-tile">${newMeaningfulForSelectedTypeCount}</span></li>`;
        comparisonSummaryText += `<li>Improved tiles (for ${selectedResolutionType}): <span class="summary-improved-res">${improvedForSelectedTypeCount}</span></li>`;
        comparisonSummaryText += `</ul>`;

        if (newMeaningfulForSelectedTypeCount > 0) {
            plotData.push({
                x: uniqueYears, y: uniqueCountries, z: zNewHighlight, type: 'heatmap',
                colorscale: [[0, 'rgba(220, 38, 38, 0.55)'], [1, 'rgba(220, 38, 38, 0.55)']],
                zmin: 4, zmax: 4, showscale: false, hoverinfo: 'skip'
            });
        }
        if (improvedForSelectedTypeCount > 0) {
            plotData.push({
                x: uniqueYears, y: uniqueCountries, z: zImprovedHighlight, type: 'heatmap',
                colorscale: [[0, 'rgba(37, 99, 235, 0.55)'], [1, 'rgba(37, 99, 235, 0.55)']],
                zmin: 5, zmax: 5, showscale: false, hoverinfo: 'skip'
            });
        }
        showMessage(comparisonSummaryText, 'comparison_summary', 'summaryPanel');

    } else if (performComparison) {
        let errorSummaryText = `<h4 class="text-xl font-semibold text-slate-800 mb-4 pb-2 border-b border-slate-300">Comparison Information</h4><ul class="space-y-1 text-sm text-slate-700">`;
        errorSummaryText += `<li>Displaying: ${selectedCurrentVersion || 'N/A'}</li>`;
        if (!selectedCurrentVersion || !selectedPreviousVersion) {
            errorSummaryText += `<li><span class="font-medium text-amber-600">Warning:</span> Please select both a current and a previous version for comparison.</li>`;
        } else if (selectedCurrentVersion === selectedPreviousVersion) {
            errorSummaryText += `<li><span class="font-medium text-amber-600">Warning:</span> Cannot compare a version to itself. Please select different versions.</li>`;
        } else {
             errorSummaryText += `<li><span class="font-medium text-slate-600">Note:</span> Comparison conditions not fully met or no comparable data.</li>`;
        }
        errorSummaryText += `</ul>`;
        showMessage(errorSummaryText, 'comparison_summary', 'summaryPanel');
    } else {
        let globalSummaryText = `<h4 class="text-xl font-semibold text-slate-800 mb-4 pb-2 border-b border-slate-300">Global Summary</h4><ul class="space-y-1 text-sm text-slate-700">`;
        globalSummaryText += `<li><strong>Displaying Version:</strong> ${versionForHeatmap}</li>`;
        globalSummaryText += `<li><strong>Region:</strong> ${region}</li>`;
        globalSummaryText += `<li><strong>Years:</strong> ${uniqueYears.length > 0 ? uniqueYears[0] + ' - ' + uniqueYears[uniqueYears.length -1] : 'N/A'} </li>`;
        globalSummaryText += `<li><strong>Total Countries in Selection:</strong> ${uniqueCountries.length}</li>`;
        globalSummaryText += `<li><strong>Total Tiles Displayed:</strong> ${totalDisplayedTiles}</li>`;
        Object.keys(levelTexts).forEach(levelKey => {
            const text = levelTexts[levelKey];
            const countVal = counts[text] || 0;
            const percentage = totalDisplayedTiles > 0 ? ((countVal / totalDisplayedTiles) * 100).toFixed(1) : "0.0";
            globalSummaryText += `<li>${text} Tiles: ${countVal} (${percentage}%)</li>`;
        });
        globalSummaryText += `</ul>`;
        showMessage(globalSummaryText, 'global_summary', 'summaryPanel');
    }

    let titleSuffix = selectedResolutionType === "Temporal" ? "Temporal Resolution" : "Spatial Resolution";
    const layout = {
        title: { text: `Best ${titleSuffix} Available - ${region} (${versionForHeatmap})`, x: 0.5, xanchor: 'center', font: { family: 'Inter, sans-serif', size: 18, color: '#1e293b'}},
        autosize: true, height: 780,
        xaxis: { title: {text: 'Year', font: {family: 'Inter, sans-serif', size: 13, color: '#334155'}}, dtick: uniqueYears.length > 20 ? Math.ceil(uniqueYears.length / 15) : 1, tickangle: uniqueYears.length > 25 ? 45 : 0, tickfont: { family: 'Inter, sans-serif', size: 11, color: '#475569'}, gridcolor: '#e2e8f0', automargin: true },
        yaxis: { title: {text: 'Country', font: {family: 'Inter, sans-serif', size: 13, color: '#334155'}}, automargin: true, tickfont: { family: 'Inter, sans-serif', size: uniqueCountries.length > 30 ? 9 : 11, color: '#475569'}, gridcolor: '#e2e8f0'},
        margin: { t: 60, b: 80, r: 20, l: uniqueCountries.length > 10 ? 150 : 100 },
        plot_bgcolor: '#ffffff', paper_bgcolor: '#ffffff', showlegend: false,
        font: { family: 'Inter, sans-serif', size: 12, color: '#334155' }
    };

    Plotly.react(heatmapDiv, plotData, layout, {responsive: true});
}

    console.error("One or more critical DOM elements are missing. Dashboard cannot initialize fully.");

}


function togglePreviousVersionControl(show) {
    if (show) {
        previousVersionControlSpan.style.display = 'flex';
        previousVersionControlSpan.classList.add('flex-col', 'sm:flex-row', 'sm:items-center', 'sm:space-x-2');
    } else {
        previousVersionControlSpan.style.display = 'none';
    }
}

function triggerUpdate() {
    if (allDataCache) {
        const selectedRegion = regionSelect.value;
        const selectedResType = resolutionTypeSelect.value;
        const performComparison = comparisonCheckbox.checked;
        const selectedCurrent = currentVersionSelect.value;
        const selectedPrevious = previousVersionSelect.value;
        updateHeatmap(selectedRegion, allDataCache, performComparison, selectedCurrent, selectedPrevious, selectedResType);
    } else {
        showMessage("Data is not loaded yet. Please wait or try refreshing.", "error");
    }
}

function initializePlot() {
    showMessage("Loading data, please wait...", "info");
    loadData().then(data => {
        hideMessage();
        allDataCache = data;
        if (!allDataCache || allDataCache.length === 0) {
            showMessage("Failed to load data or data is empty. Check 'heatmap.csv' and console. Ensure the file is accessible and you are not running into CORS issues if viewing the HTML directly from the filesystem (try a local server).", "error");
            document.getElementById('currentVersionSelect').disabled = true;
            document.getElementById('previousVersionSelect').disabled = true;
            return;
        }
        populateVersionDropdowns(allDataCache, 'currentVersionSelect', 'previousVersionSelect');
        updatePreviousVersionSelection(); // Call this after populating to set initial state
        togglePreviousVersionControl(comparisonCheckbox.checked);
        triggerUpdate(); // Initial plot
    }).catch(error => {
        console.error('Initialization failed during data load:', error);
        // The showMessage call is likely already handled in loadCSVData's error callback
    });
}

regionSelect.addEventListener('change', triggerUpdate);
resolutionTypeSelect.addEventListener('change', triggerUpdate);
comparisonCheckbox.addEventListener('change', () => {
    togglePreviousVersionControl(comparisonCheckbox.checked);
    triggerUpdate();
});
currentVersionSelect.addEventListener('change', () => {
    if (allDataCache && allDataCache.length > 0) {
        updatePreviousVersionSelection();
    }
    triggerUpdate();
});
previousVersionSelect.addEventListener('change', triggerUpdate);

initializePlot();

window.addEventListener('resize', () => {
    const heatmapDiv = document.getElementById('heatmapDiv');
    // Check if data is loaded and heatmapDiv is actually part of the document and visible
    if (allDataCache && heatmapDiv && heatmapDiv.offsetParent !== null) {
        Plotly.Plots.resize(heatmapDiv);
    }
});

});
